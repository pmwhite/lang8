# Classical higher-order logic (Church's simple type theory), written in
# Pure as natural-deduction rules: each connective's ways in and out.
# This file is the whole of what the logic assumes; every region builds on
# it with fixed symbols and hypotheses only, never more axioms.

type bool
const Trueprop : bool => prop
notation hidden Trueprop
role Trueprop Trueprop

const True : bool
const False : bool
const eq : 'a => 'a => bool
const imp : bool => bool => bool
const conj : bool => bool => bool
const disj : bool => bool => bool
const Not : bool => bool
const All : ('a => bool) => bool
const Ex : ('a => bool) => bool

notation infix 50 "=" eq
notation infixr 25 "-->" imp
notation infixr 25 "⟶" imp
notation infixr 35 "&" conj
notation infixr 35 "∧" conj
notation infixr 30 "∨" disj
notation prefix 40 "~" Not
notation prefix 40 "¬" Not
notation binder "ALL" All
notation binder "∀" All
notation binder "EX" Ex
notation binder "∃" Ex

role imp imp
role conj conj
role disj disj
role not Not
role all All
role ex Ex
role eq eq
role false False

# Equality
axiom refl "Both sides are the same": x = x
axiom subst "substitution": s = t ⟹ P s ⟹ P t
axiom ext "Show it for each input": (⋀x. f x = g x) ⟹ f = g
axiom eq_reflection: x = y ⟹ x ≡ y
axiom iff "Show both ways": (P ⟹ Q) ⟹ (Q ⟹ P) ⟹ P = Q

# Implication
axiom impI "Assume it": (P ⟹ Q) ⟹ P ⟶ Q
axiom mp "Apply it": P ⟶ Q ⟹ P ⟹ Q

# And, or
axiom conjI "Split": P ⟹ Q ⟹ P ∧ Q
axiom conjunct1 "Take the left part": P ∧ Q ⟹ P
axiom conjunct2 "Take the right part": P ∧ Q ⟹ Q
axiom disjI1 "Show the left": P ⟹ P ∨ Q
axiom disjI2 "Show the right": Q ⟹ P ∨ Q
axiom disjE "Cases": P ∨ Q ⟹ (P ⟹ R) ⟹ (Q ⟹ R) ⟹ R

# For all, there is
axiom allI "Take any _": (⋀x. P x) ⟹ ∀x. P x
axiom spec "Use with _": (∀x. P x) ⟹ P x
axiom exI "Choose _": P x ⟹ ∃x. P x
axiom exE "Get _": (∃x. P x) ⟹ (⋀x. P x ⟹ Q) ⟹ Q

# True, false, not; and the excluded middle, which makes it classical
axiom TrueI "True": True
axiom FalseE "False gives anything": False ⟹ P
axiom notI "Suppose it": (P ⟹ False) ⟹ ¬P
axiom notE "Contradiction": ¬P ⟹ P ⟹ R
axiom ccontr "Suppose not": (¬P ⟹ False) ⟹ P

# Rules used inside other moves, never offered by themselves
hide subst eq_reflection

# Rules made from the ones above, each proved here by the moves under it
# (the same moves a player makes), so they add nothing to what is assumed.
lemma sym "Flip sides": s = t ⟹ t = s
  proof:
    part "t" "s"

lemma spec_mp "Use with _": (∀x. P x ⟶ Q x) ⟹ P x ⟹ Q x
  proof:
    hyp 0 "Use with ..." with "x"
    hyp 2 "Apply it"

lemma allI_imp "Take any _": (⋀x. P x ⟹ Q x) ⟹ ∀x. P x ⟶ Q x
  proof:
    goal "Take any x"
    goal "Assume it"
    hyp 0 "Apply it"
