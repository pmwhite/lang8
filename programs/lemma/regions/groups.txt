# Groups: a set with a way to combine two of its elements, an element that
# changes nothing, and an inverse for each. Everything here is proved from
# the five rules of the rulebook, so it holds in every group.

region groups "Groups"
picture square
chapter "These five rules are all you get. Everything else, you prove."

typevar G
fix mul : G => G => G
fix e : G
fix inv : G => G

# (Not left-associative: every product is shown in brackets, since which
# way a product is grouped is what many moves here change.)
notation infix 70 "*" mul
notation infix 70 "·" mul
notation postfix 1100 "⁻¹" inv

assume assoc "Regroup": (a · b) · c = a · (b · c)
assume id_left "e on the left": e · a = a
assume id_right "e on the right": a · e = a
assume inv_left "Inverse on the left": a⁻¹ · a = e
assume inv_right "Inverse on the right": a · a⁻¹ = e

level nothing "Nothing changes"
  say: "e changes nothing it is put with. Take the e's away."
  demo: face
  word: eq
  show: (e · a) · e = a
  par: 2
  proof:
    part "(e · a) · e" "e · a"
    part "e · a" "a"

level undo "Undo"
  say: "a⁻¹ undoes a. Put the a next to its inverse."
  show: a · (a⁻¹ · b) = b
  par: 3
  proof:
    part "a · (a⁻¹ · b)" "(a · a⁻¹) · b"
    part "a · a⁻¹" "e"
    part "e · b" "b"

level id_unique "Only one e"
  say: "If f changes nothing it is put with, it is e."
  show: (∀x. f · x = x) ⟹ f = e
  par: 2
  proof:
    part "f" "f · e"
    hyp 0 "Use with ..." with "e"

level cancel "Cancel on the left"
  say: "If a·b and a·c are the same, then b and c are. Undo the a with its inverse."
  show: a · b = a · c ⟹ b = c
  par: 7
  proof:
    part "b" "e · b"
    part "e" "□⁻¹ · □" with "a"
    part "(a⁻¹ · a) · b" "a⁻¹ · (a · b)"
    part "a · b" "a · c"
    part "a⁻¹ · (a · c)" "(a⁻¹ · a) · c"
    part "a⁻¹ · a" "e"
    part "e · c" "c"

level inv_unique "Only one inverse"
  say: "If a·b is e, then b is the inverse of a: there is no other."
  show: a · b = e ⟹ b = a⁻¹
  par: 5
  proof:
    part "b" "e · b"
    part "e" "□⁻¹ · □" with "a"
    part "(a⁻¹ · a) · b" "a⁻¹ · (a · b)"
    part "a · b" "e"
    part "a⁻¹ · e" "a⁻¹"

level inv_inv "The inverse of the inverse"
  say: "Undoing an undo gets back where you were."
  demo: flip
  show: (a⁻¹)⁻¹ = a
  tools: inv_unique
  par: 3
  proof:
    goal "Flip sides"
    goal "Only one inverse"
    part "a⁻¹ · a" "e"

level inv_mul "The inverse of a product"
  say: "Undoing a·b means undoing b, then a."
  show: (a · b)⁻¹ = b⁻¹ · a⁻¹
  tools: inv_unique
  par: 7
  proof:
    goal "Flip sides"
    goal "Only one inverse"
    part "(a · b) · (b⁻¹ · a⁻¹)" "a · (b · (b⁻¹ · a⁻¹))"
    part "b · (b⁻¹ · a⁻¹)" "(b · b⁻¹) · a⁻¹"
    part "b · b⁻¹" "e"
    part "e · a⁻¹" "a⁻¹"
    part "a · a⁻¹" "e"

level cancel_right "Cancel on the right"
  say: "If b·a and c·a are the same, then b and c are."
  show: b · a = c · a ⟹ b = c
  par: 7
  proof:
    part "b" "b · e"
    part "e" "□ · □⁻¹" with "a"
    part "b · (a · a⁻¹)" "(b · a) · a⁻¹"
    part "b · a" "c · a"
    part "(c · a) · a⁻¹" "c · (a · a⁻¹)"
    part "a · a⁻¹" "e"
    part "c · e" "c"
