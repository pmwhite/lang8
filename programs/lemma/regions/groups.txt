# Groups: a set with a way to combine two of its elements, an element that
# changes nothing, and an inverse for each. Everything here is proved from
# the rules of the rulebook, so it holds in every group.
#
# It is played with beads and a tile (beads.l8): a product is a string of
# beads, an inverse a bead turned over, e no bead at all. The first levels
# move a square tile by a quarter turn and a flip (the symmetries of a
# square, computed by the two rules turn_def and flip_def); the rest are
# equations of beads, with facts as stamps.

region groups "Groups"
picture beads
chapter "A tile to turn and flip, then strings of beads. Make them match."

typevar G
fix mul : G => G => G
fix e : G
fix inv : G => G

# (Not left-associative: every product is shown in brackets, since which
# way a product is grouped is what many moves here change.)
notation infix 70 "*" mul
notation infix 70 "·" mul
notation postfix 1100 "⁻¹" inv

assume assoc "Regroup": (a · b) · c = a · (b · c)
assume id_left "e on the left": e · a = a
assume id_right "e on the right": a · e = a
assume inv_left "Inverse on the left": a⁻¹ · a = e
assume inv_right "Inverse on the right": a · a⁻¹ = e

# The square tile: its corners (top left, top right, bottom right, bottom
# left), and the two moves, as what they do to the corners.
typevar Corner
typevar Pos
typevar Move
fix sq : Corner => Corner => Corner => Corner => Pos
fix act : Move => Pos => Pos
fix turn : Move
fix flip : Move

assume turn_def "Quarter turn": act(turn, sq(a, b, c, d)) = sq(d, a, b, c)
assume flip_def "Flip": act(flip, sq(a, b, c, d)) = sq(b, a, d, c)

# What the beads do by themselves, proved here from the rulebook: an
# inverse of a string is the string turned round, every bead turned over.
lemma inv_unique "Only one inverse": a · b = e ⟹ b = a⁻¹
  proof:
    part "b" "e · b"
    part "e" "□⁻¹ · □" with "a"
    part "(a⁻¹ · a) · b" "a⁻¹ · (a · b)"
    part "a · b" "e"
    part "a⁻¹ · e" "a⁻¹"

lemma inv_inv "The inverse of the inverse": (a⁻¹)⁻¹ = a
  proof:
    goal "Flip sides"
    goal "Only one inverse"
    part "a⁻¹ · a" "e"

lemma inv_mul "The inverse of a product": (a · b)⁻¹ = b⁻¹ · a⁻¹
  proof:
    goal "Flip sides"
    goal "Only one inverse"
    part "(a · b) · (b⁻¹ · a⁻¹)" "a · (b · (b⁻¹ · a⁻¹))"
    part "b · (b⁻¹ · a⁻¹)" "(b · b⁻¹) · a⁻¹"
    part "b · b⁻¹" "e"
    part "e · a⁻¹" "a⁻¹"
    part "a · a⁻¹" "e"

lemma inv_e "e undone": e⁻¹ = e
  proof:
    goal "Flip sides"
    goal "Only one inverse"
    part "e · e" "e"

lemma turn_round "Turned round": a⁻¹ = b⁻¹ ⟹ a = b
  proof:
    part "a" "a · e"
    part "e" "□⁻¹ · □" with "b"
    part "a · (b⁻¹ · b)" "(a · b⁻¹) · b"
    part "b⁻¹" "a⁻¹"
    part "a · a⁻¹" "e"
    part "e · b" "b"

# ---- the tile ----
level quarter "A quarter turn"
  say: "Tap the tile to turn it. Make it look like its shadow."
  demo: walk
  show: ∃x. ∃y. act(y, act(x, sq(A, B, C, D))) = sq(C, D, A, B)
  par: 2
  proof:
    walk turn
    walk turn

level over "Over"
  say: "Swipe the tile sideways to flip it over. Two moves."
  demo: flipwalk
  show: ∃x. ∃y. act(y, act(x, sq(A, B, C, D))) = sq(C, B, A, D)
  par: 2
  proof:
    walk flip
    walk turn

level other_way "The other way round"
  say: "Turns go only one way. Get to a quarter turn the other way, in three moves."
  show: ∃x. ∃y. ∃z. act(z, act(y, act(x, sq(A, B, C, D)))) = sq(B, C, D, A)
  par: 3
  proof:
    walk flip
    walk turn
    walk flip

level shortcut "A short cut"
  say: "A flip and three turns got here. Find a way in two."
  show: ∃x. ∃y. act(y, act(x, sq(A, B, C, D))) = act(turn, act(turn, act(turn, act(flip, sq(A, B, C, D)))))
  par: 2
  proof:
    walk turn
    walk flip

# ---- beads ----
level pop "Pop"
  say: "A bead and its turned-over twin, side by side, pop. Tap one, then the other."
  demo: pop
  show: (a · b) · (b⁻¹ · a⁻¹) = c · c⁻¹
  par: 3
  proof:
    pop top 1
    pop top 0
    pop bottom 0

level stamp "A stamp"
  say: "The card says a b is the same as c. Tap one side of it, then beads like it."
  demo: stamp
  show: a · b = c ⟹ a⁻¹ · c = b
  par: 2
  proof:
    stamp 0 face 1 top 1 1
    pop top 0

level out_of_nothing "Out of nothing"
  say: "A bead and its twin can come out of nothing: tap a bead below, then a gap."
  demo: pair
  show: a · b = a · c ⟹ b = c
  par: 3
  proof:
    pair "a⁻¹" top 0
    stamp 0 face 0 top 1 2
    pop top 0

level turned "Turned around"
  say: "a and b can swap. Can their twins? Swipe the strings sideways to turn them around."
  demo: turn
  show: a · b = b · a ⟹ a⁻¹ · b⁻¹ = b⁻¹ · a⁻¹
  par: 2
  proof:
    turn
    stamp 0 face 1 top 0 2

level does_nothing "Does nothing"
  say: "If f in front of any beads leaves them as they were, f is nothing. X on a card is any beads at all, even none."
  show: (∀x. f · x = x) ⟹ f = e
  par: 1
  proof:
    stamp 0 face 0 top 0 1

level twice "Twice is nothing"
  say: "Any beads twice over are nothing, and nothing is any beads twice over. Show a b is b a."
  show: (∀x. x · x = e) ⟹ a · b = b · a
  par: 3
  proof:
    stamp 0 face 1 bottom 0 0 with "a"
    stamp 0 face 1 bottom 4 0 with "b"
    stamp 0 face 0 bottom 1 4

# How the tile's moves and turning round read in a proof written out.
says turn_def "After a quarter turn, we must show {now}."
says flip_def "After a flip, we must show {now}."
says turn_round "Turning both sides round, it is enough to show {now}."
