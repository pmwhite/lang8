# The rulebook. Calculate leans on all of it: a field, ordered, with |x|
# the larger of x and −x, and numerals sums of 1; the natural numbers in
# order, with max.
assume add_assoc: (x + y) + z = x + (y + z)
assume add_comm: x + y = y + x
assume add_zero: x + 0 = x
assume add_neg: x + -x = 0
assume sub_def: x - y = x + -y
assume mul_assoc: (x · y) · z = x · (y · z)
assume mul_comm: x · y = y · x
assume mul_one: x · 1 = x
assume distrib: x · (y + z) = x · y + x · z
assume div_def: ¬(y = 0) ⟹ (x / y) · y = x
assume zero_ne_one: ¬(0 = 1)
assume two: 2 = 1 + 1
assume le_def: (x ≤ y) = (x < y ∨ x = y)
assume lt_irrefl: ¬(x < x)
assume lt_trans: x < y ⟹ y < z ⟹ x < z
assume lt_total: x < y ∨ x = y ∨ y < x
assume add_lt: x < y ⟹ x + z < y + z
assume mul_pos: 0 < x ⟹ 0 < y ⟹ 0 < x · y
assume abs_pos: 0 ≤ x ⟹ |x| = x
assume abs_neg: x < 0 ⟹ |x| = -x
assume nle_refl: (m::ℕ) ≤ m
assume nle_trans: (k::ℕ) ≤ m ⟹ m ≤ n ⟹ k ≤ n
assume nle_total: (m::ℕ) ≤ n ∨ n ≤ m
assume max_left: (n::ℕ) ≤ m ⟹ max m n = m
assume max_right: ¬((n::ℕ) ≤ m) ⟹ max m n = n

# (The rulebook is Calculate's to use: not offered as moves of its own.)
hide add_assoc add_comm add_zero add_neg sub_def mul_assoc mul_comm mul_one distrib div_def
  zero_ne_one two le_def lt_irrefl lt_trans lt_total add_lt mul_pos abs_pos abs_neg nle_refl
  nle_trans nle_total max_left max_right

# What it means to settle down to L: however close you ask (ε), from some
# point on (N) the sequence stays that close.
define tendsto a L "Open the definition" = ∀ε>0. ∃N. ∀n≥N. |a n − L| < ε
notation infix 45 "→" tendsto

level const_limit "Standing still"
  say: "A sequence that is c every time settles down to c."
  show: (λn. c) → c
  kit: 0
  par: 4
  proof:
    part "(λn. c) → c" "Open the definition"
    goal "Take any ε"
    goal "Choose ..." with "0"
    goal "Take any n"

level shifted "Shifted"
  say: "If a settles down to A, then a + 1 settles down to A + 1. Ask a for the same closeness."
  show: a → A ⟹ (λn. a n + 1) → A + 1
  par: 8
  proof:
    part "(λn. a_n + 1) → A + 1" "Open the definition"
    goal "Take any ε"
    part "a → A" in 0 "Open the definition"
    hyp 0 "Use with ..." with "ε"
    hyp 2 "Get N"
    goal "Choose ..." with "N"
    goal "Take any n"
    hyp 2 "Use with ..." with "n"

level doubled "Doubled"
  say: "If a settles down to A, then 2·a settles down to 2·A. Ask a for half the closeness."
  show: a → A ⟹ (λn. 2 · a n) → 2 · A
  kit: x / 2
  par: 8
  proof:
    part "(λn. 2 · a_n) → 2 · A" "Open the definition"
    goal "Take any ε"
    part "a → A" in 0 "Open the definition"
    hyp 0 "Use with ..." with "ε / 2"
    hyp 2 "Get N"
    goal "Choose ..." with "N"
    goal "Take any n"
    hyp 2 "Use with ..." with "n"

level sum_limit "Adding limits"
  say: "If a settles down to A, and b to B, then a + b settles down to A + B. Ask each for half the closeness."
  show: a → A ⟹ b → B ⟹ (λn. a n + b n) → A + B
  kit: x / 2, max(x, y)
  par: 12
  proof:
    part "(λn. a_n + b_n) → A + B" "Open the definition"
    goal "Take any ε"
    part "a → A" in 0 "Open the definition"
    hyp 0 "Use with ..." with "ε / 2"
    hyp 3 "Get N"
    part "b → B" in 1 "Open the definition"
    hyp 1 "Use with ..." with "ε / 2"
    hyp 4 "Get N₁"
    goal "Choose ..." with "max(N, N₁)"
    goal "Take any n"
    hyp 3 "Use with ..." with "n"
    hyp 4 "Use with ..." with "n"

level limit_unique "Only one limit"
  say: "A sequence cannot settle down to two different numbers. Suppose it did: ask for less than half the gap."
  show: a → A ⟹ a → B ⟹ A = B
  kit: |x − y| / 2, max(x, y)
  par: 9
  proof:
    goal "Suppose not"
    part "a → A" in 0 "Open the definition"
    part "a → B" in 1 "Open the definition"
    hyp 0 "Use with ..." with "|A − B| / 2"
    hyp 3 "Get N"
    hyp 1 "Use with ..." with "|A − B| / 2"
    hyp 4 "Get N₁"
    hyp 3 "Use with ..." with "max(N, N₁)"
    hyp 4 "Use with ..." with "max(N, N₁)"
