# The first region: the logic's own moves, on statements about anything.

region logic "Logic"

typevar U

level and_swap "And, the other way round"
  say: "Two things are true. Show they are true in the other order."
  show: P ∧ Q ⟹ Q ∧ P
  par: 3
  proof:
    goal "Split"
    hyp 0 "Take the right part"
    hyp 0 "Take the left part"

level chain "A chain of ifs"
  say: "If P gives Q, and Q gives R, then P gives R."
  show: (P ⟶ Q) ⟹ (Q ⟶ R) ⟹ (P ⟶ R)
  par: 3
  proof:
    goal "Assume it"
    hyp 1 "Apply it"
    hyp 0 "Apply it"

level swap_quantifiers "Someone for everyone"
  say: "If one x is related to every y, then every y has an x related to it."
  show: (∃x:U. ∀y:U. R x y) ⟹ ∀y. ∃x. R x y
  par: 4
  proof:
    goal "Take any y"
    hyp 0 "Get x"
    goal "Choose ..." with "x"
    hyp 0 "Use with y"
