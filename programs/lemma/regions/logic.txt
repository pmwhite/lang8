# The first region: the logic's own moves, on statements about anything.
# It is the tutorial of the game's language: each level brings one word
# (a key on a fact, a key on the goal, a fact put onto the goal, a name
# dropped into a notch), and the later ones put them together.

region logic "Logic"

typevar U

level take_left "Take what you need"
  say: "You know P and Q. Show P."
  demo: key
  word: and
  show: P ∧ Q ⟹ P
  par: 1
  proof:
    open hyp 0

level both "Both at once"
  say: "You know P, and you know Q. Show both."
  demo: goalkey
  show: P ⟹ Q ⟹ P ∧ Q
  par: 1
  proof:
    open goal

level and_swap "And, the other way round"
  say: "Two things are true. Show they are true in the other order."
  show: P ∧ Q ⟹ Q ∧ P
  par: 2
  proof:
    open hyp 0
    open goal

level suppose "Suppose"
  say: "To show P gives Q, suppose P. Here Q is known already."
  word: if
  show: Q ⟹ (P ⟶ Q)
  par: 1
  proof:
    open goal

level modus_ponens "If, then"
  say: "If P gives Q, and P is true, then Q is."
  demo: drag
  show: (P ⟶ Q) ⟹ P ⟹ Q
  par: 1
  proof:
    apply hyp 0

level chain "A chain of ifs"
  say: "If P gives Q, and Q gives R, then P gives R."
  show: (P ⟶ Q) ⟹ (Q ⟶ R) ⟹ (P ⟶ R)
  par: 3
  proof:
    open goal
    apply hyp 1
    apply hyp 0

level any_one "Any one at all"
  say: "Everything has P and Q. So everything has P."
  demo: token
  word: all
  show: (∀x:U. P x ∧ Q x) ⟹ ∀y. P y
  par: 3
  proof:
    open goal
    drop "y" hyp 0
    open hyp 1

level for_every "For every one"
  say: "Every x with P gives Q, and a has P. Use the for-all with a."
  show: (∀x:U. P x ⟶ Q) ⟹ P a ⟹ Q
  par: 2
  proof:
    drop "a" hyp 0
    apply hyp 2

level show_one "Show me one"
  say: "If a has P, then something has P."
  demo: notch
  word: ex
  show: P a ⟹ ∃x:U. P x
  par: 1
  proof:
    drop "a" goal

level one_of_them "One of them"
  say: "Something has P and Q. So something has Q."
  demo: some
  show: (∃x:U. P x ∧ Q x) ⟹ ∃y. Q y
  par: 3
  proof:
    open hyp 0
    open hyp 0
    drop "x" goal

level swap_quantifiers "Someone for everyone"
  say: "If one x is related to every y, then every y has an x related to it."
  show: (∃x:U. ∀y:U. R x y) ⟹ ∀y. ∃x. R x y
  par: 4
  proof:
    open goal
    open hyp 0
    drop "x" goal
    drop "y" hyp 0
